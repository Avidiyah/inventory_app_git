# Runtime dependencies for the Inventory Management backend.
#
# Pinned to the verified working environment (pip freeze of backend/venv).
# Curated to direct runtime needs; pip resolves the remaining transitive
# packages. Dev/test-only tools live in requirements-dev.txt.
#
# NOTE on pyzbar: it wraps the native `zbar` library. The wheel bundles the
# DLLs, but on Windows they require the Visual C++ 2013 Redistributable
# (msvcr120.dll); without it the import fails with a missing libiconv.dll.
# See docs/current-state.md.

fastapi==0.136.3
uvicorn==0.48.0
# uvicorn is installed WITHOUT [standard], so it ships no WebSocket
# protocol implementation. Without this package uvicorn answers every
# upgrade request with "Unsupported upgrade request" and closes.
#
# This cannot be caught by the test suite: Starlette's TestClient drives
# `websocket_connect` against the ASGI app directly and never imports a
# protocol library, so the socket tests pass either way. See
# tests/test_realtime_dependency.py, which exists solely to guard this.
#
# 15.0.1 is the version verified against uvicorn 0.48.0 in this environment.
# It is a compatibility pin, not a claim about the newest available release.
# uvicorn 0.48.0's handshake still
# imports the deprecated `websockets.legacy` module, so a future websockets
# major that drops it will break that handshake silently -- bumping this pin
# requires re-verifying a real server connection, not just a green suite.
websockets==15.0.1
# 1.2.1 carried 2 advisories and sits under every route in the app. Bumped to
# the minimum fixing version rather than latest (1.6.0) -- three extra minors
# of ASGI-layer change buy no additional CVE coverage. fastapi 0.136.3 declares
# `starlette>=0.46.0` with no upper bound, so this does not fight its pin.
starlette==1.3.1
pydantic==2.13.4
SQLAlchemy==2.0.50
alembic==1.18.4
psycopg==3.3.4
psycopg-binary==3.3.4
python-dotenv==1.2.2
python-multipart==0.0.32
pyzbar==0.1.9
# Parses attacker-supplied image data on the barcode upload path
# (routers/barcodes.py -> services/barcodes.py) before pyzbar sees it, so its
# advisories were the most exposed of the pip-audit baseline. 12.2.0 carried
# the bulk of the 23 CVEs; 12.3.0 is both the fix and the current release.
pillow==12.3.0
# Web Push (RFC 8030 / 8291 / 8292). The de-facto Python implementation, from
# the same org as the reference VAPID libraries. Synchronous and `requests`-
# based, which is a feature here rather than a limitation: every route handler
# in this app is a sync `def` in a threadpool, so delivery matches the app's
# existing concurrency model instead of fighting it.
#
# Verified at install: `webpush()` is a plain `def` returning a
# requests.Response. It also pulls in aiohttp as a hard dependency for an
# async path we do not use -- so aiohttp appears in the tree, and in the
# pip-audit surface, without any of our code importing it.
#
# 2.4.0 is current (2026-08-06). Pinned exactly: PyPI flags this a "Critical
# Project" maintained by one person, so a version bump is a supply-chain event
# to choose deliberately rather than inherit.
pywebpush==2.4.0
